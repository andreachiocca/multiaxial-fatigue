% registry: MGSE_YU
\methodhead{MGSE\_YU}{Yu, Zhu, Liu and Liu (2017): modified generalized strain energy}\label{sheet:MGSE_YU}

\block{Source} Z.~Y.~Yu, S.~P.~Zhu, Q.~Liu, Y.~Liu, \emph{Materials} 10(8) (2017) 923
\cite{yu2017}, Eq.~(11).

\block{Idea and scope} An energy-type critical-plane parameter without material-specific
constants: the product of the maximum shear stress and the shear strain amplitude is added to the
product of the maximum normal stress and the normal strain amplitude on the same plane. The mean
stress enters through the two maxima. Proposed for multiaxial fatigue lives of ductile and brittle
metals under proportional and non-proportional loading.

\block{Inputs}
\begin{center}\small
\begin{tabular}{@{}llp{100mm}@{}}
\toprule
Symbol & Unit & Meaning\\\midrule

$\tau_{\max}$ & MPa & largest magnitude of the shear stress vector on the plane over the cycle, mean included\\
$\Delta\gamma$, $\gamma_a$ & -- & range and amplitude ($\Delta\gamma/2$) of the engineering shear strain on the plane\\
$\sigma_{n,m}$, $\sigma_{n,a}$ & MPa & mean and amplitude of the normal stress on the plane, Eqs.~\eqref{eq:plane}, \eqref{eq:scalar}\\
$\sigma_{n,\max}$ & MPa & largest normal stress on the plane over the cycle, clipped at zero\\

$\theta$, $c=\cos^2\theta$ & rad, -- & angle between the plane normal and the load axis in uniaxial tension\\
$\sigma_a$, $\tau_a$ & MPa & fully reversed axial and shear stress amplitudes (closed forms)\\
$\mathrm{MGSE}$, $\mathrm{DP}$ & MPa & damage parameter (MJ/m$^3$)\\
\bottomrule
\end{tabular}
\end{center}

\block{Formulation} On a critical plane close to the plane of maximum shear strain (Eq.~11),
\begin{equation}
\mathrm{MGSE} = \tau_{\max}\,\frac{\Delta\gamma}{2} + \sigma_{n,\max}\,\frac{\Delta\varepsilon_n}{2} .
\label{eq:mgseyu-main}
\end{equation}
The repository evaluates, on every plane,
\begin{equation}
\mathrm{DP} = \tau_{\max}\,\gamma_a + \sigma_{n,\max}\,\varepsilon_{n,a},\qquad
\sigma_{n,\max} = \max(\sigma_{n,m}+\sigma_{n,a},\,0) .
\label{eq:mgseyu-plane}
\end{equation}
\emph{Critical plane}: the plane of maximum DP (a repository reading, see Repository choices);
the selection metric is DP itself, so ties are immaterial and \texttt{MGSE\_YU\_ext} equals
\texttt{MGSE\_YU}. \emph{Shear amplitude}: default \texttt{np\_shear\_rule = "rss"}, the MCE
amplitude $\sqrt{a^2+b^2}$ of the elliptical shear path with semiaxes $a$, $b$; the user may select
the longest chord (equal to the MCC for one harmonic) globally (\texttt{docs/shear-amplitude.md},
``Selection for every method'').

With elastic strains, fully reversed tension on a plane with $c\ge\nu/(1+\nu)$ gives
\begin{equation}
\mathrm{DP}(\theta) = \frac{\sigma_a^2}{E}\bigl[(2+\nu)\,c-(1+\nu)\,c^2\bigr],\qquad
\mathrm{DP}_{\max} = \frac{\sigma_a^2}{E}\,\frac{(2+\nu)^2}{4(1+\nu)},
\label{eq:mgseyu-tension}
\end{equation}
and fully reversed torsion $\mathrm{DP}=2(1+\nu)\tau_a^2/E$. Equal DP at the two limits implies
$\tm/\fm=(2+\nu)/[2\sqrt2(1+\nu)]\approx0.625$ for $\nu=0.3$.

\block{Constants and calibration} None: the parameter has no criterion constant. Only the master
curve is fitted.

\block{Life} The repository uses the master curve on DP, Eq.~\eqref{eq:mastercurve}, fitted on the
uniaxial $R=-1$ design set. The guide does not record the paper's own life relation.

\block{Repository choices}
\begin{itemize}
\item Maximum-damage critical plane: the paper chooses a plane close to the maximum shear plane;
  the repository maximises the full MGSE over all candidate planes. This is not an exact
  reproduction of a restricted plane search.
\item Non-proportional shear range by the default \texttt{rss} rule; $\tau_{\max}$ from 720 time
  samples of the harmonic cycle (endpoint maximum when the plane method is called directly).
\item $\sigma_{n,\max}<0$ is clipped to $0$.
\item Elastic strains: with them \texttt{MGSE\_YU} reduces to the same scalar expression as elastic
  GSE (sheet~\ref{sheet:GSE}).
\item Life from the shared master curve.
\end{itemize}

\block{Worked example} Synthetic material, $E=200\,000\MPa$, $\nu=0.3$, zero mean: tension
$\sigma_a=300\MPa$ gives $\mathrm{DP}=0.457788\MPa$, Eq.~\eqref{eq:mgseyu-tension}
(\texttt{\_ext} identical); torsion $\tau_a=200\MPa$ gives $0.520000\MPa$; tension $200\MPa$ with
torsion $150\MPa$ at $90^\circ$ gives $0.492500\MPa$ (\texttt{\_ext} identical).

al.'s restricted search near the maximum shear plane and their experimental data are not
reproduced in \texttt{tests/methods/}.
