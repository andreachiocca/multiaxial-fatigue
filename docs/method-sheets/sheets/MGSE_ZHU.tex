% registry: MGSE_ZHU
\methodhead{MGSE\_ZHU}{Zhu, Yu, Correia, De Jesus and Berto (2018): weighted generalized strain energy}\label{sheet:MGSE_ZHU}

\block{Source} S.-P.~Zhu, Z.-Y.~Yu, J.~Correia, A.~De~Jesus, F.~Berto, \emph{Int.\ J.\ Fatigue} 113
(2018) 213--229 \cite{zhu2018}, Eqs.~(7)--(12), pp.~7--8; Section~4.1, pp.~12--13.

\block{Idea and scope} A generalisation of MGSE (sheet~\ref{sheet:MGSE_YU}) to ductile and brittle
metals: the normal strain-energy term on the critical plane is weighted by a material constant
$k_\mathrm{MGSE}$. The mean stress enters through the maximum shear and normal stresses. Proposed
for multiaxial fatigue lives of both ductile and brittle metals.

\block{Formulation} The paper writes (Eqs.~7--12)
\begin{equation}
\mathrm{MGSE} = \tau_{\max}\,\frac{\Delta\gamma}{2} + k_\mathrm{MGSE}\,\sigma_{n,\max}\,\frac{\Delta\varepsilon_n}{2} .
\label{eq:mgsezhu-main}
\end{equation}
The repository evaluates, on every plane,
\begin{equation}
\mathrm{DP} = \tau_{\max}\,\gamma_a + k_\mathrm{MGSE}\,\sigma_{n,\max}\,\varepsilon_{n,a},\qquad
\sigma_{n,\max} = \max(\sigma_{n,m}+\sigma_{n,a},\,0) .
\label{eq:mgsezhu-plane}
\end{equation}
\emph{Critical plane}: the plane of maximum DP (provisional, see Repository choices); the selection
metric is DP itself, so ties are immaterial and \texttt{MGSE\_ZHU\_ext} equals \texttt{MGSE\_ZHU}.
\emph{Shear amplitude}: default \texttt{np\_shear\_rule = "rss"}, the MCE amplitude
$\sqrt{a^2+b^2}$ of the elliptical shear path; the longest chord (MCC) can be selected globally
(\texttt{docs/shear-amplitude.md}, ``Selection for every method'').

\block{Constants and calibration} \emph{Paper}: Eq.~(12) expresses $k_\mathrm{MGSE}$ as a function
of life from the axial and torsional strain--life constants; Section~4.1 averages it over a
finite-life range (5000--50\,000 cycles for most metals) and recommends $k_\mathrm{MGSE}=1$ when
fatigue properties are unavailable. \emph{Repository} (\texttt{multifat-pipeline}), when
$k_\mathrm{MGSE}$ is in neither the workbook nor \texttt{--constants}:
\begin{enumerate}
\item if Basquin curves are identifiable on the $R=-1$ tension and torsion failures, the elastic
  terms only of Eq.~(12), averaged over 200 log-spaced lives $N\in[10^4,\,2\cdot10^6]$:
  \begin{equation}
  k_\mathrm{MGSE}(N) = \frac{\dfrac{\tau_f'^2}{G}(2N)^{2b_0} - \dfrac{1+\nu}{2}\,\dfrac{\sigma_f'^2}{E}(2N)^{2b}}
  {\dfrac{\sigma_f'^2}{2E}(2N)^{2b}} ;
  \label{eq:mgsezhu-k}
  \end{equation}
\item otherwise, minimum $\ln N_f$ scatter of the $R=-1$ tension and torsion failures around one
  master curve, $k_\mathrm{MGSE}\in[0,5]$;
\item on fatigue-limit datasets, $k_\mathrm{MGSE}\in[0,5]$ equalises the axial and torsional
  limit DP.
\end{enumerate}
The tests behind $k_\mathrm{MGSE}$ are design rows (\texttt{Flag = 0}). In single-model runs
(\texttt{multifat-run}) without a value, $k_\mathrm{MGSE}=1$ with a warning.

\block{Life} The repository uses the master curve on DP, Eq.~\eqref{eq:mastercurve}, fitted on the
uniaxial $R=-1$ design set, instead of a strain--life curve.

\block{Repository choices}
\begin{itemize}
\item Critical plane: maximum of the full weighted MGSE over all planes. The plane rule and the
  weighting remain provisional: the audit records that the original equation and plane rule were
  not independently reproduced from the accessible article.
\item Non-proportional shear range by the default \texttt{rss} rule; $\tau_{\max}$ from 720 time
  samples of the harmonic cycle (endpoint maximum outside the plane preparation).
\item $\sigma_{n,\max}<0$ is clipped to $0$.
\item $k_\mathrm{MGSE}$ optional, default 1.0 with a warning; Eq.~\eqref{eq:mgsezhu-k} keeps the
  elastic terms of Eq.~(12) only and averages over $10^4$--$2\cdot10^6$ cycles, not the paper's
  5000--50\,000.
\item Elastic strains; life from the shared master curve.
\end{itemize}

\block{Worked example} Synthetic material, $E=200\,000\MPa$, $\nu=0.3$, $k_\mathrm{MGSE}=0.8$,
zero mean. Tension $\sigma_a=300\MPa$: on the plane $c^*=\min\bigl(1,A_z/(2B_z)\bigr)$ with
$A_z=2(1+\nu)-k\nu$, $B_z=(2-k)(1+\nu)$, $\mathrm{DP}=\sigma_a^2(A_zc^*-B_zc^{*2})/E=0.401654\MPa$
(\texttt{\_ext} identical). Torsion $\tau_a=200\MPa$: $\mathrm{DP}=2(1+\nu)\tau_a^2/E=0.520000\MPa$.
Eq.~\eqref{eq:mgsezhu-k} with $\sigma_f'=1200\MPa$, $b=-0.08$, $\tau_f'=700\MPa$, $b_0=-0.075$ gives
$k_\mathrm{MGSE}=0.7063$.

\block{Verification} \texttt{tests/test\_method\_audit.py} checks the closed-form plane value;
\texttt{tests/test\_pipeline.py} checks that the $k_\mathrm{MGSE}$ calibration does not depend on
which other models are requested. The paper's life benchmarks are not reproduced.
